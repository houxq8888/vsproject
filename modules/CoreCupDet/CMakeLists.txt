set(LIBRARY_VERSION_MAJOR 1)
set(LIBRARY_VERSION_MINOR 0)
set(LIBRARY_VERSION_PATCH 0)


HG_CMAKE_ADD_MODULE(CoreCupDet inc src ../../)


HG_CMAKE_MODULE_DEPEND_PUB( 
    HGBaseCommon
    HGSvcImageAlgorithm
    ${ZXING_LIBRARY}
    ${OpenCV_LIBRARIES}
    ${LOG4CPLUS_LIBRARIES}  
)
if (PLATFORM STREQUAL "x64")
HG_CMAKE_ADD_SHARED(HGCoreCupDet inc src ../../)

HG_CMAKE_MODULE_DEPEND(
    HGBaseCommon
    HGSvcImageAlgorithm
    ${ZXING_LIBRARY}
    ${OpenCV_LIBRARIES}
    ${LOG4CPLUS_LIBRARIES}  
)
elseif (PLATFORM STREQUAL "win32")
HG_CMAKE_ADD_STATIC(HGCoreCupDet inc src)
HG_CMAKE_MODULE_DEPEND(
    HGBaseCommon    
    HGSvcImageAlgorithm
    ${OpenCV_LIBRARIES}
)

endif()

set_target_properties(HGCoreCupDet PROPERTIES 
    VERSION ${LIBRARY_VERSION_MAJOR}.${LIBRARY_VERSION_MINOR}.${LIBRARY_VERSION_PATCH}
    SOVERSION ${LIBRARY_VERSION_MAJOR}
)
